\documentclass{article}
\usepackage[textwidth=6cm]{geometry}% just for the preview

\usepackage{tikz}
\usetikzlibrary{lindenmayersystems}

\usepackage[active,tightpage]{preview}
\PreviewEnvironment{center}
\setlength\PreviewBorder{5pt}%

\pgfdeclarelindenmayersystem{A}{
	\rule{F -> FF[+F][-F]}
}

\pgfdeclarelindenmayersystem{B}{
	\rule{F -> ffF[++FF][--FF]}
}

\pgfdeclarelindenmayersystem{C}{
	\symbol{G}{\pgflsystemdrawforward}
	\rule{F -> F[+F][-F]FG[+F][-F]FG}
}

\pgfdeclarelindenmayersystem{D}{
	\symbol{G}{\pgflsystemdrawforward}
	\symbol{H}{\pgflsystemdrawforward}
	\rule{F -> H[+HG][-HG]G}
	\rule{G -> HF}
}

% Classic Koch Snowflake Contour
\pgfdeclarelindenmayersystem{E}{
	\rule{F -> F-F++F-F}
}

% Hexagonal Web (Koch Island)
\pgfdeclarelindenmayersystem{F}{
	\rule{F -> F+F--F+F}
}

% Sierpinski Arrowhead
\pgfdeclarelindenmayersystem{G}{
	\rule{A -> B-A-B}
	\rule{B -> A+B+A}
}

\tikzset{
	type/.style={l-system={#1, axiom=F,order=3,step=4pt,angle=60},
		blue, opacity=0.4, line width=.5mm, line cap=round
	},
}

\newcommand\drawsnowflake[2][scale=0.2]{
	\tikz[#1]
	\foreach \a in {0,60,...,300}  {
		\draw[rotate=\a,#2] l-system;
	};
}

\begin{document}

\begin{center}
	% Type A
	\foreach \width in {.2,.4,...,.8}
	{  \drawsnowflake[scale=0.3]{type=A, line width=\width mm} }

	% Type A (90 degrees)
	\foreach \width in {.2,.4,...,.8}
	{  \drawsnowflake[scale=0.38]{type=A, l-system={angle=90}, line
	width=\width mm} }

	% Type B
	\foreach \width in {.2,.4,...,.8}
	{  \drawsnowflake[scale=0.3]{type=B, line width=\width mm} }

	% Type B (30 degrees)
	\foreach \width in {.2,.4,...,.8}
	{  \drawsnowflake{type=B, l-system={angle=30}, line width=\width mm} }

	% Type C
	\drawsnowflake[scale=0.24]{type=C, l-system={order=2}, line width=0.2mm}
	\drawsnowflake[scale=0.25]{type=C, l-system={order=2}, line width=0.4mm}
	\drawsnowflake[scale=0.25]{type=C, l-system={order=2,axiom=fF}, line
	width=0.2mm}
	\drawsnowflake[scale=0.32]{type=C,
	l-system={order=2,axiom=---fff+++F}, line width=0.2mm}

	% Type D
	\drawsnowflake[scale=0.36]{type=D,
	l-system={order=4,angle=60,axiom=GF}, line width=0.7mm}
	\drawsnowflake[scale=0.36]{type=D,
	l-system={order=4,angle=60,axiom=GfF}, line width=0.7mm}
	\drawsnowflake[scale=0.36]{type=D,
	l-system={order=4,angle=60,axiom=FG}, line width=0.7mm}
	\drawsnowflake[scale=0.36]{type=D,
	l-system={order=4,angle=60,axiom=FfG}, line width=0.7mm}

	% Type E Classic Koch Snowflake
	\foreach \width in {.3,.5,.7,.9}
	{ \drawsnowflake[scale=0.14]{type=E, l-system={axiom=F++F++F,
	order=3}, line width=\width mm} }

	% Type F Hexagonal Web
	\foreach \width in {.3,.5,.7,.9}
	{ \drawsnowflake[scale=0.14]{type=F, l-system={order=3}, line
	width=\width mm} }

	% Type G Sierpinski Arrowhead
	\drawsnowflake[scale=0.14]{type=G, l-system={axiom=A, order=4}, line
	width=0.5mm}
	\drawsnowflake[scale=0.14]{type=G, l-system={axiom=B, order=4}, line
	width=0.5mm}
\end{center}

\end{document}
