\documentclass[varwidth=true, border=5pt]{standalone}

\usepackage{tikz}
\usetikzlibrary{calc,through,intersections}
\usepackage{ifthen}

% Definition of the command showing the nine points
\newcommand\ninepointscircle[3]{%
\begin{tikzpicture}
	% Vertices of the triangle
	\coordinate[label=above:$A$] (A) at (#1);
	\coordinate[label=left:$B$]  (B) at (#2);
	\coordinate[label=right:$C$] (C) at (#3);

	% Middles of sides
	\coordinate[label=below:$A'$] (A') at ($(B)!.5!(C)$);
	\coordinate[label=right:$B'$] (B') at ($(A)!.5!(C)$);
	\coordinate[label=left:$C'$]  (C') at ($(B)!.5!(A)$);

	% Projections of vertices on opposite sides
	\coordinate[label=below:$H_A$] (HA) at ($(B)!(A)!(C)$);
	\coordinate[label=45:$H_B$]    (HB) at ($(A)!(B)!(C)$);
	\coordinate[label=135:$H_C$]   (HC) at ($(B)!(C)!(A)$);

	% Drawing the triangle
	\draw[name path=triangle] (A) -- (B) -- (C) -- cycle;

	% Medians
	\draw[color=red,name path=ma] (A)--(A');
	\draw[color=red,name path=mb] (B)--(B');
	\draw[color=red,name path=mc] (C)--(C');

	% Altitudes
	\draw[fill=blue,color=blue,name path=ha] (A) -- (HA);
	\draw[fill=blue,color=blue,name path=hb] (B) -- (HB);
	\draw[fill=blue,color=blue,name path=hc] (C) -- (HC);

	% Right angle marks for altitudes
	% \tkzRightAngle[color=blue](B/HC/C,C/HA/A,A/HB/B)
	\draw[color=blue] ($(HA)!0.15!(A)$) -- ($(HA)!0.15!(A)!0.15!90:(A)$) -- ($(HA)!0.15!(C)$);
	\draw[color=blue] ($(HB)!0.15!(B)$) -- ($(HB)!0.15!(B)!0.15!90:(B)$) -- ($(HB)!0.15!(C)$);
	\draw[color=blue] ($(HC)!0.15!(C)$) -- ($(HC)!0.15!(C)!0.15!90:(C)$) -- ($(HC)!0.15!(A)$);

	% Drawing perpendicular bisectors
	% \tkzMathLength(A,HA)
	% \path [name path=C1] (A') -- ($(A')!\tkzMathLen pt!90:(C)$);
	% \tkzMathLength(B,HB)
	% \path [name path=C2] (B') -- ($(B')!\tkzMathLen pt!90:(A)$);
	% \tkzMathLength(C,HC)
	% \path [name path=C3] (C') -- ($(C')!\tkzMathLen pt!90:(B)$);
	% \draw[color=green,name intersections={of=triangle and C1,name=ch1,sort by=C1,total=\t}] 
	%    (ch1-\t)--(A') coordinate (OA') at (ch1-\t);
	% \draw[color=green,name intersections={of=triangle and C2,name=ch2,sort by=C2,total=\t}] 
	%    (ch2-\t)--(B') coordinate (OB') at (ch2-\t);
	% \draw[color=green,name intersections={of=triangle and C3,name=ch3,sort by=C3,total=\t}] 
	%    (ch3-\t)--(C') coordinate (OC') at (ch3-\t);
	\coordinate (perpA) at ($(A')!2!90:(B)$);
	\coordinate (perpB) at ($(B')!2!90:(C)$);
	\coordinate (perpC) at ($(C')!2!90:(A)$);
	
	\draw[color=green] (A') -- (perpA);
	\draw[color=green] (B') -- (perpB);
	\draw[color=green] (C') -- (perpC);

	% Find circumcenter
	\coordinate (O) at (intersection of A'--perpA and B'--perpB);

	% Right angle marks for perpendicular bisectors
	% \tkzRightAngle[color=green](B/C'/OC',C/A'/OA',A/B'/OB')
	\draw[color=green] ($(A')!0.15!(O)$) -- ($(A')!0.15!(O)!0.15!90:(O)$) -- ($(A')!0.15!(perpA)$);
	\draw[color=green] ($(B')!0.15!(O)$) -- ($(B')!0.15!(O)!0.15!90:(O)$) -- ($(B')!0.15!(perpB)$);
	\draw[color=green] ($(C')!0.15!(O)$) -- ($(C')!0.15!(O)!0.15!90:(O)$) -- ($(C')!0.15!(perpC)$);

	% Orthocenter
	\coordinate[label=45:$H$] (H) at (intersection of A--HA and B--HB);
	
	% Centroid
	\coordinate[label=0:$G$] (G) at (intersection of A--A' and B--B');

	% Draw points
	\draw[fill=red] (G) circle (.8pt);
	\draw[fill=blue] (H) circle (.8pt);
	\draw[fill=green] (O) circle (.8pt);

	% Nine points circle center (midpoint of OH)
	\coordinate (N) at ($(H)!.5!(O)$);

	% Draw Euler line
	\draw[color=orange] (G) -- ($(G)!3!(O)$);
	\draw[color=orange] (G) -- ($(G)!3!(H)$);
	\node[color=orange] at ($(G)!1.5!(O)$) [right] {Euler's line};

	% Marking midpoints of segments from vertices to orthocenter
	\coordinate (I) at ($(A)!.5!(H)$);
	\coordinate (J) at ($(B)!.5!(H)$);
	\coordinate (K) at ($(C)!.5!(H)$);
	
	\draw[fill=black] (I) circle (.8pt) node[anchor=south west] {$I$};
	\draw[fill=black] (J) circle (.8pt) node[below] {$J$};
	\draw[fill=black] (K) circle (.8pt) node[right] {$K$};

	% Markings for equal segments
	\draw ($(A)!0.3!(I)$) -- ($(A)!0.7!(I)$);
	\draw ($(H)!0.3!(I)$) -- ($(H)!0.7!(I)$);
	\draw ($(B)!0.3!(J)$) -- ($(B)!0.7!(J)$);
	\draw ($(H)!0.3!(J)$) -- ($(H)!0.7!(J)$);
	\draw ($(C)!0.3!(K)$) -- ($(C)!0.7!(K)$);
	\draw ($(H)!0.3!(K)$) -- ($(H)!0.7!(K)$);

	% Mark the nine points
	\draw[fill=black] (A') circle (.8pt);
	\draw[fill=black] (B') circle (.8pt);
	\draw[fill=black] (C') circle (.8pt);
	\draw[fill=black] (HA) circle (.8pt);
	\draw[fill=black] (HB) circle (.8pt);
	\draw[fill=black] (HC) circle (.8pt);
	\draw[fill=black] (I) circle (.8pt);
	\draw[fill=black] (J) circle (.8pt);
	\draw[fill=black] (K) circle (.8pt);

	% Drawing the nine points circle
	% \node[fill=lightgray,opacity=.2,draw,circle through=(C'),
	%    label=90:$\mathcal{C}$] at (N) {};
	\node[fill=lightgray, opacity=.2, draw, circle through=(C')] at (N) {};
\end{tikzpicture}
}

\begin{document}

\end{document}
