% !TeX program = lualatex
% !TeX encoding = UTF-8

\documentclass[x11names, border=5pt]{standalone}

\usepackage{tikz}
\usetikzlibrary{calc,decorations.markings}
\usepackage{amsmath}

% commands
\def\pr#1{\directlua{tex.print(#1)}}

\begin{document}

% parameters
\definecolor{sand}{RGB}{193,154,107}
\def\cols{20}
\def\rows{40}
\def\squareunit{.35}
\pgfmathsetmacro\rmaxparticle{.1}
\def\beforelight{5}

\begin{tikzpicture}[x = \squareunit cm, y = \squareunit cm, line width = 2pt]
	
	% Light path before particles cloud
	\draw[red,
		decoration = {markings,
				mark = at position 0.5 with {\arrow[red]{latex}}},
		postaction = {decorate}] (-\beforelight,{\rows*\squareunit/2})--++
	(\beforelight,0) node[midway, above, black] {$i_0$};

	% Light path through particles cloud (using shading)
	\begin{scope}
		\clip (0,0) rectangle (\cols*\squareunit,\rows*\squareunit);
		\shade[left color=red, right color=red!20!white] 
			(0,0) rectangle (\cols*\squareunit,\rows*\squareunit);
		\draw[red, line width=2pt] 
			(0,{\rows*\squareunit/2}) -- (\cols*\squareunit,{\rows*\squareunit/2});
	\end{scope}

	% Light path after particles cloud
	\draw[red!20!white,
		decoration = {markings,
				mark = at position 0.5 with {\arrow[red!20!white]{latex}}},
		postaction = {decorate}] ({\cols*\squareunit},{\rows*\squareunit/2})--++
	(\beforelight,0) node[midway, above, black] {$i$};

	% Particles cloud
	\directlua{
		math.randomseed(os.time())
		for i = 1, \cols do
			for j = 1, \rows do
				local radius = \rmaxparticle * math.random()
				local l = \squareunit - 2 * radius
				local x = (i-1) * \squareunit + radius + l * math.random()
				local y = (j-1) * \squareunit + radius + l * math.random()
				tex.print(string.format("\\fill[sand] (%.3f,%.3f) circle[radius=%.3f];", x, y, radius))
			end
		end
	}

	% Length quoting
	\draw[|<->|, > = latex, line width = .8pt] 
		($(0,\rows*\squareunit)+(0,1)$)--++
		(\cols*\squareunit,0) node[midway, above] {$l$};

	% Lambert-Beer law
	\node[anchor = north, inner sep = 1ex] at (current bounding box.south) 
		{$\dfrac{i}{i_0}=\exp(-\gamma l)$};
\end{tikzpicture}
\end{document}
