% !TeX program = lualatex
% !TeX encoding = UTF-8

\documentclass[border=10pt]{standalone}

\usepackage{xcolor}
\usepackage[
	hmargin = 2.4cm,
	vmargin = 3cm
]{geometry}
\usepackage[
	figureposition = bottom
]{caption}
\usepackage{tikz}
\usepackage{tikz-3dplot}
\usepackage{amsmath}

% Setup of caption
\DeclareCaptionLabelSeparator{adjustment}{:\quad}
\captionsetup{
	font = small,
	labelfont = sc,
	labelsep = adjustment,
	width = 0.7\textwidth
}

\pagestyle{empty}

\begin{document}
% Set 3D viewing angle
\tdplotsetmaincoords{70}{110}
\begin{tikzpicture}[tdplot_main_coords, scale=1.2]
	% Parameters
	\def\R{4}       % Major radius of core
	\def\r{1.5}     % Minor radius of core
	\def\rWire{1.65} % Wire radius (slightly larger than tube)
	\def\Np{40}     % Number of primary windings
	\def\Ns{80}     % Number of secondary windings

	% Draw Torus core Wireframe
	% Cross-sections (rings)
	\foreach \p in {0, 20, ..., 340} {
		\draw[cyan!50!black, thin, opacity=0.4]
		plot[variable=\t, domain=0:360, samples=24]
		({(\R+\r*cos(\t))*cos(\p)}, {(\R+\r*cos(\t))*sin(\p)}, {\r*sin(\t)});
	}
	\foreach \t in {0, 45, ..., 315} {
		\draw[cyan!50!black, thin, opacity=0.4]
		plot[variable=\p, domain=0:360, samples=48]
		({(\R+\r*cos(\t))*cos(\p)}, {(\R+\r*cos(\t))*sin(\p)}, {\r*sin(\t)});
	}

	% Draw Primary Coil, bottom half
	\foreach \i in {0, 1, ..., 29} {
		\pgfmathsetmacro{\thetaStart}{180 + \i * 6}
		\draw[red!80!black, thick, line cap=round]
		plot[variable=\t, domain=0:360, samples=12]
		({(\R+\rWire*cos(\t))*cos(\thetaStart + \t*6/360)},
			{\rWire*sin(\t)},
		{(\R+\rWire*cos(\t))*sin(\thetaStart + \t*6/360)});
	}

	% Draw Secondary Coil, top half
	\foreach \i in {0, 1, ..., 29} {
		\pgfmathsetmacro{\thetaStart}{0 + \i * 6}
		\draw[blue!80!black, thick, line cap=round]
		plot[variable=\t, domain=0:360, samples=12]
		({(\R+\rWire*cos(\t))*cos(\thetaStart + \t*6/360)},
			{\rWire*sin(\t)},
		{(\R+\rWire*cos(\t))*sin(\thetaStart + \t*6/360)});
	}

	% Magnetic flux arrow through the core
	\draw[dashed, purple!80!black, very thick, ->
		, decoration={markings,
		mark=at position 0.5 with {\arrow[scale=1.5]{Stealth}}},
	postaction={decorate}]
	plot[variable=\p, domain=0:300, samples=120]
	({(\R+0.05)*cos(\p)}, {(\R+0.05)*sin(\p)}, {0});

	% Primary labels
	\node[text=red!80!black, font=\bfseries, fill=white, inner sep=1pt,
	opacity=0.9, text opacity=1]
	at ({(\R+1.5)*cos(255)}, {(\R+1.5)*sin(255)}, 0) {Primary};
	\node[text=red!80!black, font=\small]
	at ({(\R+1.5)*cos(235)}, {(\R+1.5)*sin(235)}, 0) {\(N_p = 40\)};

	% Secondary labels
	\node[text=blue!80!black, font=\bfseries, fill=white, inner sep=1pt,
	opacity=0.9, text opacity=1]
	at ({(\R+1.5)*cos(45)}, {(\R+1.5)*sin(45)}, 0) {Secondary};
	\node[text=blue!80!black, font=\small]
	at ({(\R+1.5)*cos(65)}, {(\R+1.5)*sin(65)}, 0) {\(N_s = 80\)};

	% Flux label
	\node[text=purple!80!black, font=\small, fill=white, inner sep=1pt,
	opacity=0.9, text opacity=1]
	at ({(\R-0.05)*cos(315)}, {(\R-0.05)*sin(315)}, 0) {\(\Phi\)};

	% Voltage / current callouts
	% Primary side (bottom-left of torus)
	\node[red!80!black, font=\small] (Ip)  at ({(\R+2.5)*cos(200)},
	{(\R+2.5)*sin(200)}, 0) {\(U_p\)};
	\node[red!80!black, font=\small] (Ipl) at ({(\R+2.5)*cos(170)},
	{(\R+2.5)*sin(170)}, 0) {\(I_p\)};

	% Secondary side (top-right of torus)
	\node[blue!80!black, font=\small] (Us)  at ({(\R+2.5)*cos(20)},
	{(\R+2.5)*sin(20)}, 0) {\(U_s\)};
	\node[blue!80!black, font=\small] (Usl) at ({(\R+2.5)*cos(50)},
	{(\R+2.5)*sin(50)}, 0) {\(I_s\)};
\end{tikzpicture}

\end{document}
