% !TeX program = lualatex
% !TeX encoding = UTF-8

\documentclass[x11names, border=5pt]{standalone}

\usepackage{tikz}
\usetikzlibrary{calc, decorations.markings}
\usepackage{amsmath}

\begin{document}

\definecolor{sand}{RGB}{193,154,107}
\def\cols{20}
\def\rows{40}
\def\squareunit{0.35}
\def\rmaxparticle{0.1}
\def\beforelight{5}
\def\tauVal{2}

\begin{tikzpicture}[x=\squareunit cm, y=\squareunit cm, line width=2pt]

	% incoming beam
	\draw[red,
		decoration={markings, mark=at position 0.5 with {\arrow[red]{latex}}},
	postaction={decorate}]
	(-\beforelight,{\rows*\squareunit/2}) -- ++(\beforelight,0)
	node[midway, above, black] {$I_0$};

	% beam through cloud: EXPONENTIAL fade
	\foreach \x in {0,0.05,...,\cols} {
		\pgfmathsetmacro{\opacity}{exp(-\tauVal*\x/\cols)}
		\draw[red, opacity=\opacity, line width=2pt]
		(\x*\squareunit,{\rows*\squareunit/2}) --
		({(\x+0.05)*\squareunit},{\rows*\squareunit/2});
	}

	% outgoing beam
	\draw[red!13.5!white,
		decoration={markings, mark=at position 0.5 with
		{\arrow[red!13.5!white]{latex}}},
	postaction={decorate}]
	({\cols*\squareunit},{\rows*\squareunit/2}) -- ++(\beforelight,0)
	node[midway, above, black] {$I = I_0 e^{-\tau}$};

	% Monte Carlo rays + cloud
	\directlua{
		config = config or {}
		config.cols         = \number\cols\relax
		config.rows         = \number\rows\relax
		config.squareunit   = \squareunit
		config.rmaxparticle = \rmaxparticle
		config.tau          = 2.0
		dofile("particles.lua")
	}

	% length quoting
	\draw[|<->|, > = latex, line width=.8pt]
	($(0,\rows*\squareunit)+(0,1)$) -- ++(\cols*\squareunit,0)
	node[midway, above] {$l$};

	% Physics annotation block
	\node[anchor=north, align=left, font=\small,
		text=black!80, fill=white, fill opacity=0.85,
	text opacity=1, inner sep=1ex, rounded corners=2pt]
	at ({1}, 3)
	{$\gamma = n\,\sigma$ \;\;(density $\times$ cross-section)\\[1pt]
		$\tau = \gamma l =
		\pgfmathparse{\tauVal}\pgfmathprintnumber{\pgfmathresult}$ (optical
		depth)\\[1pt]
	$\dfrac{I}{I_0} = e^{-\tau} = e^{-\gamma l} \approx 0.135$};

	% Inset: I/I0 vs x
	\begin{scope}[shift={(\cols*\squareunit + 1.5, -1.8)},
		x=1.6cm, y=1.6cm]
		\draw[->, thin] (0,0) -- (2.4,0) node[right, font=\footnotesize] {$x$};
		\draw[->, thin] (0,0) -- (0,1.2) node[above, font=\footnotesize]
		{$\frac{I}{I_0}$};
		\draw[red, thick, domain=0:2.2, samples=60]
		plot (\x, {exp(-\tauVal*\x/2.2)});
		\draw[dashed, thin] (2.2, 0) -- (2.2, {exp(-\tauVal)})
		node[pos=0.5, right, font=\scriptsize] {$e^{-\tau}$};
		\node[font=\scriptsize] at (1.5, 0.9) {$e^{-\gamma x}$};
	\end{scope}
\end{tikzpicture}
\end{document}
