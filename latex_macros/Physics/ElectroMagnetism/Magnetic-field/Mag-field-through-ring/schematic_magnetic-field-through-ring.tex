% !TeX program = lualatex
% !TeX encoding = UTF-8

\documentclass[border=10pt]{standalone}

\usepackage{amsmath}
\usepackage{tikz}
\usepackage{tikz-3dplot}
\usetikzlibrary{arrows.meta, decorations.markings, positioning}

\tikzset{
	>={Stealth[scale=1.2]},
	midarrow/.style={
		decoration={markings, mark=at position 0.53 with
		{\arrow{Stealth[scale=1.2]}}},
		postaction={decorate}
	}
}

% Set the 3D viewing angle (elevation, azimuth)
\tdplotsetmaincoords{65}{120}

\begin{document}

% First Picture: 3D Magnetic Lenses / Coils
\begin{tikzpicture}[tdplot_main_coords, scale=1.5]
	\def\R{1.5}
	\def\zoffset{1.5}

	% Top Coil (Blue-ish)
	\draw[thick, blue!80!black] 
		plot[variable=\t,domain=0:360,samples=100] ({\R*cos(\t)}, {\R*sin(\t)}, \zoffset);
	\draw[thick, blue!80!black] 
		plot[variable=\t,domain=0:360,samples=100] ({\R*cos(\t)}, {\R*sin(\t)}, \zoffset+0.1);

	% Bottom Coil (Red-ish)
	\draw[thick, red!80!black] 
		plot[variable=\t,domain=0:360,samples=100] ({\R*cos(\t)}, {\R*sin(\t)}, -\zoffset);
	\draw[thick, red!80!black] 
		plot[variable=\t,domain=0:360,samples=100] ({\R*cos(\t)}, {\R*sin(\t)}, -\zoffset-0.1);

	% Cylinder Wireframe (To give depth)
	\foreach \t in {0,30,...,330} {
		\draw[blue!20, thin] ({\R*cos(\t)}, {\R*sin(\t)}, \zoffset) -- ({\R*cos(\t)}, {\R*sin(\t)}, -\zoffset);
	}

	% Particle Beams (Orange) - 3D Curves passing through
	\draw[thick, orange!90!black, ->] 
		plot[variable=\z,domain=-\zoffset-1:\zoffset+1,samples=50] ({-0.6}, {0}, {\z});
	\draw[thick, orange!90!black, ->] 
		plot[variable=\z,domain=-\zoffset-1:\zoffset+1,samples=50] ({0}, {0}, {\z});
	\draw[thick, orange!90!black, ->] 
		plot[variable=\z,domain=-\zoffset-1:\zoffset+1,samples=50] ({0.6}, {0}, {\z});

	% Current I (Purple)
	\draw[thick, purple!80!black, ->, samples=100] 
		plot[variable=\t,domain=0:360] ({\R*cos(\t)}, {\R*sin(\t)}, -\zoffset-0.5);
	\node[font=\small, text=purple!80!black, fill=white, inner sep=2pt, opacity=0.8, text opacity=1] 
		at (0, 1, -\zoffset-1) {\(\boldsymbol{I}\)};
\end{tikzpicture}

\vspace{1cm}

% Second Picture: 3D Tokamak / Plasma Torus
\begin{tikzpicture}[tdplot_main_coords, scale=1.2]
	\def\R{2.5}
	\def\r{0.8}
	\def\rp{0.4}

	% Outer Vessel (3D Wireframe)
	\foreach \u in {0,30,...,330} {
		\draw[gray!40, thin] 
			plot[variable=\v,domain=0:360,samples=50] 
			({(\R+\r*cos(\v))*cos(\u)}, {(\R+\r*cos(\v))*sin(\u)}, {\r*sin(\v)});
	}
	\foreach \v in {0,30,...,330} {
		\draw[gray!40, thin] 
			plot[variable=\u,domain=0:360,samples=50] 
			({(\R+\r*cos(\v))*cos(\u)}, {(\R+\r*cos(\v))*sin(\u)}, {\r*sin(\v)});
	}

	% Inner Plasma Torus (Orange Mesh)
	\foreach \u in {0,30,...,330} {
		\draw[orange!70!black, thick] 
			plot[variable=\v,domain=0:360,samples=50] 
			({(\R+\rp*cos(\v))*cos(\u)}, {(\R+\rp*cos(\v))*sin(\u)}, {\rp*sin(\v)});
	}
	\foreach \v in {0,30,...,330} {
		\draw[orange!70!black, thick] 
			plot[variable=\u,domain=0:360,samples=50] 
			({(\R+\rp*cos(\v))*cos(\u)}, {(\R+\rp*cos(\v))*sin(\u)}, {\rp*sin(\v)});
	}

	% Toroidal Field B_theta (Helical)
	\draw[thick, blue!80!black, ->, samples=200] 
		plot[variable=\t,domain=0:360] 
		({(\R+0.5*cos(3*\t))*cos(\t)}, {(\R+0.5*cos(3*\t))*sin(\t)}, {0.5*sin(3*\t)});
	\node[text=blue!80!black, font=\small, fill=white, inner sep=2pt, opacity=0.8, text opacity=1] 
		at (4, 0, 0) {\(\boldsymbol{B_\theta}\)};

	% Poloidal Field B_phi (Loop around cross-section)
	\draw[thick, red!80!black, ->, samples=100] 
		plot[variable=\t,domain=0:360] 
		({(\R+0.5*cos(\t))*cos(180)}, {(\R+0.5*cos(\t))*sin(180)}, {0.5*sin(\t)});
	\node[text=red!80!black, font=\small, fill=white, inner sep=2pt, opacity=0.8, text opacity=1] 
		at (-\R, 0, 1.5) {\(\boldsymbol{B_\phi}\)};

	% Plasma Current I_phi (Large loop along the torus)
	\draw[thick, green!60!black, ->, samples=100] 
		plot[variable=\t,domain=0:360] 
		({(\R+0.5*cos(0))*cos(\t)}, {(\R+0.5*cos(0))*sin(\t)}, {0});
	\node[text=green!60!black, font=\small, fill=white, inner sep=2pt, opacity=0.8, text opacity=1] 
		at (0, \R+1, 0) {\(\boldsymbol{I_\phi}\)};
\end{tikzpicture}

\end{document}
